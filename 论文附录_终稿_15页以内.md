### B.1  问题一：单日储能调度 MILP

设 \(T=144\)，时段长度 \(\Delta t=1/6\) h。原始功率数据先按 \(E=P\Delta t\) 转换为电量。决策变量为购电量 \(x_t\)、充电量 \(c_t\)、放电量 \(d_t\)、储电量 \(E_t\)、弃光量 \(s_t\) 及充放电状态 \(z_t\in\{0,1\}\)。目标函数为全天购电费用最小：

\[
\min \sum_{t=1}^{T}\pi_t x_t.
\]

```python
# 摘自：问题一/code/first_question_core.py

def build_energy_series(price, load_kw, pv_kw, delta_t_h=DT_H):
    load_energy = load_kw * delta_t_h
    pv_energy = pv_kw * delta_t_h
    return EnergySeries(
        price_yuan_per_kwh=price,
        load_energy_kwh=load_energy,
        pv_energy_kwh=pv_energy,
        net_energy_kwh=load_energy - pv_energy,
    )


def build_milp_model(series, storage):
    T = series.time_count
    x_slice = slice(0, T)
    c_slice = slice(T, 2 * T)
    d_slice = slice(2 * T, 3 * T)
    e_slice = slice(3 * T, 4 * T)
    s_slice = slice(4 * T, 5 * T)
    z_slice = slice(5 * T, 6 * T)
    variable_count = 6 * T
    M = storage.power_kw * DT_H

    objective = np.zeros(variable_count)
    objective[x_slice] = series.price_yuan_per_kwh
    integrality = np.zeros(variable_count, dtype=int)
    integrality[z_slice] = 1

    lower = np.zeros(variable_count)
    upper = np.full(variable_count, np.inf)
    upper[c_slice] = M
    upper[d_slice] = M
    lower[e_slice] = storage.soc_min_kwh
    upper[e_slice] = storage.soc_max_kwh
    upper[s_slice] = series.pv_energy_kwh
    upper[z_slice] = 1.0
    final_e = 3 * T + T - 1
    lower[final_e] = upper[final_e] = storage.initial_kwh

    equality = lil_matrix((2 * T, variable_count))
    equality_rhs = np.zeros(2 * T)
    inequality = lil_matrix((2 * T, variable_count))
    inequality_rhs = np.zeros(2 * T)

    for t in range(T):
        x, c, d, e, s, z = (
            t,
            T + t,
            2 * T + t,
            3 * T + t,
            4 * T + t,
            5 * T + t,
        )
        # 功率平衡：购电+光伏+放电=负荷+充电+弃光。
        equality[t, [x, d, c, s]] = [1.0, 1.0, -1.0, -1.0]
        equality_rhs[t] = series.net_energy_kwh[t]

        # SOC递推：E_t=E_(t-1)+eta*c_t-d_t/eta。
        row = T + t
        equality[row, e] = 1.0
        equality[row, c] = -storage.eta_charge
        equality[row, d] = 1.0 / storage.eta_discharge
        if t == 0:
            equality_rhs[row] = storage.initial_kwh
        else:
            equality[row, e - 1] = -1.0

        # 充放电互斥：c_t<=M*z_t，d_t<=M*(1-z_t)。
        inequality[t, [c, z]] = [1.0, -M]
        row = T + t
        inequality[row, [d, z]] = [1.0, M]
        inequality_rhs[row] = M

    return MilpModel(
        objective=objective,
        integrality=integrality,
        lower_bounds=lower,
        upper_bounds=upper,
        equality_matrix=equality.tocsr(),
        equality_rhs=equality_rhs,
        inequality_matrix=inequality.tocsr(),
        inequality_rhs=inequality_rhs,
        time_count=T,
        slices={
            "grid": x_slice,
            "charge": c_slice,
            "discharge": d_slice,
            "soc": e_slice,
            "curtail": s_slice,
            "binary": z_slice,
        },
    )


def solve_daily_dispatch_milp(series, storage):
    model = build_milp_model(series, storage)
    result = milp(
        c=model.objective,
        integrality=model.integrality,
        bounds=Bounds(model.lower_bounds, model.upper_bounds),
        constraints=[
            LinearConstraint(
                model.equality_matrix,
                model.equality_rhs,
                model.equality_rhs,
            ),
            LinearConstraint(
                model.inequality_matrix,
                np.full(len(model.inequality_rhs), -np.inf),
                model.inequality_rhs,
            ),
        ],
        options={"time_limit": 300.0, "mip_rel_gap": 1e-9},
    )
    if not result.success:
        raise RuntimeError(result.message)
    return extract_dispatch_solution(series, storage, model, result.x)
```

### B.2  问题二：配对情景与两阶段随机规划

负荷预测误差和光伏预测误差按同一历史日配对，以保留两类误差的相关性。第 \(d\) 天只使用此前 `lookback_days` 个历史日的残差，情景等概率。

```python
# 摘自：问题二/problem2_stochastic.py

def generate_historical_scenarios(
    load_actual_kwh,
    pv_actual_kwh,
    reference_load_kwh,
    reference_pv_kwh,
    *,
    n_scenarios=5,
    lookback_days=30,
    error_scale=1.0,
):
    load_point, pv_point, _ = build_point_forecasts(
        load_actual_kwh,
        pv_actual_kwh,
        reference_load_kwh,
        reference_pv_kwh,
        lookback_days=lookback_days,
    )
    load_error = load_actual_kwh - load_point
    pv_error = pv_actual_kwh - pv_point
    load_scenarios = np.empty((DAYS, n_scenarios, PERIODS_PER_DAY))
    pv_scenarios = np.empty_like(load_scenarios)

    for day in range(DAYS):
        if day == 0:
            history = np.array([-1])
        else:
            start = max(0, day - lookback_days)
            history = np.arange(start, day)
        selected = (
            history
            if n_scenarios == 1
            else history[
                np.linspace(
                    0,
                    len(history) - 1,
                    n_scenarios,
                ).round().astype(int)
            ]
        )
        for k, error_day in enumerate(selected):
            if error_day < 0:
                load_residual = np.zeros(PERIODS_PER_DAY)
                pv_residual = np.zeros(PERIODS_PER_DAY)
            else:
                load_residual = load_error[error_day]
                pv_residual = pv_error[error_day]
            load_scenarios[day, k] = np.maximum(
                0.0,
                load_point[day] + error_scale * load_residual,
            )
            pv_scenarios[day, k] = np.maximum(
                0.0,
                pv_point[day] + error_scale * pv_residual,
            )

    probabilities = np.full(
        (DAYS, n_scenarios),
        1.0 / n_scenarios,
    )
    return load_scenarios, pv_scenarios, probabilities
```

计划购电量 \(X_t\)、充电量 \(c_t\)、放电量 \(d_t\)、SOC \(E_t\) 和状态 \(z_t\) 在全部情景间共享；紧急购电量 \(e_t^\omega\) 和弃光量 \(s_t^\omega\) 按情景追索。情景电能平衡与第一阶段 SOC 递推为

\[
X_t+e_t^\omega+G_t^\omega+d_t=E_t^{\mathrm{L},\omega}+c_t+s_t^\omega,
\]

\[
E_t=E_{t-1}+\eta c_t-\frac{d_t}{\eta}.
\]

```python
# 摘自：问题二/problem2_stochastic.py

def build_stochastic_model(
    load_scenarios_kwh,
    pv_scenarios_kwh,
    scenario_probabilities,
    price_144_yuan_per_kwh,
    storage,
    *,
    emergency_multiplier=5.0,
    initial_soc_kwh=None,
    terminal_soc_value_yuan_per_kwh=0.0,
    relax_binary=False,
):
    days = load_scenarios_kwh.shape[0]
    S = load_scenarios_kwh.shape[1]
    n = days * PERIODS_PER_DAY
    price_all = np.tile(price_144_yuan_per_kwh, days)
    initial_soc_kwh = (
        storage.initial_kwh
        if initial_soc_kwh is None
        else initial_soc_kwh
    )
    M = storage.power_kw * DT_H

    x_slice = slice(0, n)
    c_slice = slice(n, 2 * n)
    d_slice = slice(2 * n, 3 * n)
    e_slice = slice(3 * n, 4 * n)
    z_slice = slice(4 * n, 5 * n)
    emergency_slice = slice(5 * n, 5 * n + S * n)
    curtail_slice = slice(5 * n + S * n, 5 * n + 2 * S * n)
    variable_count = curtail_slice.stop

    objective = np.zeros(variable_count)
    objective[x_slice] = price_all
    objective[e_slice.stop - 1] -= terminal_soc_value_yuan_per_kwh
    probability_flat = scenario_probabilities.T.reshape(-1)
    objective[emergency_slice] = (
        emergency_multiplier
        * np.tile(price_all, S)
        * np.repeat(probability_flat, PERIODS_PER_DAY)
    )

    lower = np.zeros(variable_count)
    upper = np.full(variable_count, np.inf)
    upper[c_slice] = M
    upper[d_slice] = M
    lower[e_slice] = storage.soc_min_kwh
    upper[e_slice] = storage.soc_max_kwh
    upper[z_slice] = 1.0
    upper[curtail_slice] = pv_scenarios_kwh.transpose(1, 0, 2).reshape(-1)

    # 按情景组装平衡、SOC递推和充放电互斥约束。
    balance = lil_matrix((S * n, variable_count))
    soc = lil_matrix((n, variable_count))
    mutual = lil_matrix((2 * n, variable_count))
    for k in range(S):
        base = k * n
        for t in range(n):
            balance[base + t, t] = 1.0
            balance[base + t, emergency_slice.start + base + t] = 1.0
            balance[base + t, d_slice.start + t] = 1.0
            balance[base + t, c_slice.start + t] = -1.0
            balance[base + t, curtail_slice.start + base + t] = -1.0
    for t in range(n):
        soc[t, e_slice.start + t] = 1.0
        soc[t, c_slice.start + t] = -storage.efficiency
        soc[t, d_slice.start + t] = 1.0 / storage.efficiency
        if t > 0:
            soc[t, e_slice.start + t - 1] = -1.0
        mutual[t, c_slice.start + t] = 1.0
        mutual[t, z_slice.start + t] = -M
        mutual[n + t, d_slice.start + t] = 1.0
        mutual[n + t, z_slice.start + t] = M

    balance_rhs = (
        load_scenarios_kwh.transpose(1, 0, 2)
        - pv_scenarios_kwh.transpose(1, 0, 2)
    ).reshape(-1)
    soc_rhs = np.zeros(n)
    soc_rhs[0] = initial_soc_kwh

    integrality = None if relax_binary else np.zeros(variable_count, dtype=int)
    if integrality is not None:
        integrality[z_slice] = 1

    return StochasticMILPModel(
        objective=objective,
        integrality=integrality,
        bounds=Bounds(lower, upper),
        constraints=[
            LinearConstraint(balance.tocsr(), balance_rhs, balance_rhs),
            LinearConstraint(soc.tocsr(), soc_rhs, soc_rhs),
            LinearConstraint(
                mutual.tocsr(),
                np.full(2 * n, -np.inf),
                np.concatenate([np.zeros(n), np.full(n, M)]),
            ),
        ],
        variable_slices={
            "x": x_slice,
            "c": c_slice,
            "d": d_slice,
            "E": e_slice,
            "z": z_slice,
            "e": emergency_slice,
            "s": curtail_slice,
        },
        variable_count=variable_count,
        constraint_dimensions={
            "情景电能平衡": S * n,
            "第一阶段SOC递推": n,
            "充放电互斥": 2 * n,
        },
    )
```

实际执行不把计划充放电量直接固定，而是根据当前真实净负荷、SOC 和未来费用函数动态决定动作。未来费用以 61 点 SOC 网格离散，并对历史情景费用作 10% 截尾平均，以降低极端路径的影响。

```python
# 摘自：问题二/problem2_adaptive.py

def _future_value_tables(
    load_scenarios_kwh,
    pv_scenarios_kwh,
    planned_kwh,
    price_144_yuan_per_kwh,
    storage,
    *,
    terminal_soc_value_yuan_per_kwh,
    soc_grid_points,
    emergency_multiplier=5.0,
    trim_fraction=0.10,
):
    S, T = load_scenarios_kwh.shape
    soc_grid = np.linspace(
        storage.soc_min_kwh,
        storage.soc_max_kwh,
        soc_grid_points,
    )
    residual = (
        load_scenarios_kwh
        - pv_scenarios_kwh
        - planned_kwh[None, :]
    )
    value = np.repeat(
        (-terminal_soc_value_yuan_per_kwh * soc_grid)[None, :],
        S,
        axis=0,
    )
    mean_future_value = np.empty((T + 1, soc_grid_points))
    trim_count = min(
        S // 2,
        max(0, int(round(S * trim_fraction))),
    )

    def robust_mean(values):
        if trim_count <= 0:
            return np.mean(values, axis=0)
        ordered = np.sort(values, axis=0)
        return np.mean(ordered[trim_count:-trim_count], axis=0)

    mean_future_value[T] = robust_mean(value)
    M = storage.power_kw * DT_H

    for t in range(T - 1, -1, -1):
        for k in range(S):
            next_value = value[k]
            r = float(residual[k, t])
            if r <= 0.0:
                max_increase = min(
                    -r * storage.efficiency,
                    storage.efficiency * M,
                )
                target_soc = np.minimum(
                    soc_grid + max_increase,
                    storage.soc_max_kwh,
                )
                value[k] = np.interp(target_soc, soc_grid, next_value)
                continue

            max_discharge = min(r, M)
            linear_cost = (
                emergency_multiplier
                * price_144_yuan_per_kwh[t]
                * storage.efficiency
            )
            candidate = next_value + linear_cost * soc_grid
            current = np.empty(soc_grid_points)
            queue = deque()
            for index, current_soc in enumerate(soc_grid):
                feasible_discharge = min(
                    max_discharge,
                    storage.efficiency
                    * (current_soc - storage.soc_min_kwh),
                )
                lower_soc = (
                    current_soc
                    - feasible_discharge / storage.efficiency
                )
                lower_index = int(
                    np.searchsorted(soc_grid, lower_soc, side="left")
                )
                while queue and candidate[queue[-1]] >= candidate[index]:
                    queue.pop()
                queue.append(index)
                while queue and queue[0] < lower_index:
                    queue.popleft()
                current[index] = (
                    emergency_multiplier
                    * price_144_yuan_per_kwh[t]
                    * r
                    + candidate[queue[0]]
                    - linear_cost * current_soc
                )
            value[k] = current
        mean_future_value[t] = robust_mean(value)

    return soc_grid, mean_future_value
```

### B.3  问题三：因果情景与滚动调整

问题三显式约束信息时效性。负荷基准仅使用当前日之前最多 4 个同星期日的实际负荷，不足时回退到此前 7 天均值；光伏情景以当前发布时刻的预报为基准，叠加相同发布时刻、相同提前期的历史预报误差。

```python
def build_causal_load_forecast(data, current_date, fallback_profile_kwh):
    history = data[data["日期"].dt.date < current_date]
    if not history.empty:
        dates = sorted(history["日期"].dt.date.unique())
        same_weekday = [
            value
            for value in dates
            if value.weekday() == current_date.weekday()
        ][-4:]
        if same_weekday:
            selected = history[
                history["日期"].dt.date.isin(same_weekday)
            ]
            profile = (
                selected.groupby("时段序号")["小区负载电量_kWh"]
                .mean()
                .sort_index()
            )
            if len(profile) == T:
                return np.maximum(profile.to_numpy(), 0.0)

        selected = history[
            history["日期"].dt.date.isin(dates[-7:])
        ]
        profile = (
            selected.groupby("时段序号")["小区负载电量_kWh"]
            .mean()
            .sort_index()
        )
        if len(profile) == T:
            return np.maximum(profile.to_numpy(), 0.0)

    fallback = np.asarray(fallback_profile_kwh, dtype=float)
    return np.maximum(fallback, 0.0)

            data,
            forecasts,
            error_dates,
            start_hour,
        )
        current_load = np.vstack(
            [
                np.maximum(
                    0.0,
                    point_load[start:] + load_errors[k, start:],
                )
                for k in range(scenario_count)
            ]
        )
        current_pv = np.vstack(
            [
                np.maximum(
                    0.0,
                    forecast_scale * (nominal_pv + pv_errors[k]),
                )
                for k in range(scenario_count)
            ]
        )

        load_parts = [current_load]
        pv_parts = [current_pv]
        price_parts = [price_yuan_per_kwh[start:]]
        for _ in range(1, window_days):
            load_parts.append(
                np.vstack(
                    [
                        np.maximum(
                            0.0,
                            future_load + load_errors[k],
                        )
                        for k in range(scenario_count)
                    ]
                )
            )
            pv_parts.append(
                np.vstack(
                    [
                        np.maximum(
                            0.0,
                            forecast_scale * (future_pv + pv_errors[k]),
                        )
                        for k in range(scenario_count)
                    ]
                )
            )
            price_parts.append(price_yuan_per_kwh)

        windows[start_hour] = {
            "load_kwh": np.concatenate(load_parts, axis=1),
            "pv_kwh": np.concatenate(pv_parts, axis=1) * DT_H,
            "price_yuan_per_kwh": np.concatenate(price_parts),
            "probabilities": probabilities.copy(),
            "current_intervals": length,
            "selected_error_dates": tuple(error_dates),
        }
    return windows
```

0:00 形成计划购电量 \(g_t\) 并锁定；6:00、12:00、18:00 只允许修改尚未执行时段的调整购电量 \(q_t\)。正负偏差线性化为

\[
q_t-g_t=\Delta_t^+-\Delta_t^-,\qquad
\Delta_t^+,\Delta_t^-\ge 0.
\]

调整费用为 \(1.5\pi_t\Delta_t^+ +0.5\pi_t\Delta_t^-\)，紧急购电费为 \(5\pi_t e_t^F\)。

```python
# 摘自：问题三/problem3_algorithm.py

def solve_flexible_purchase_stage(
    load_scenarios_kwh,
    pv_scenarios_kwh,
    scenario_probabilities,
    price_yuan_per_kwh,
    storage,
    initial_soc_kwh,
    *,
    fixed_purchase=None,
    plan_purchase=None,
    adjustment_mode=False,
    current_intervals=None,
    scenario_price_yuan_per_kwh=None,
    time_limit_s=60.0,
):
    S, horizon = load_scenarios_kwh.shape
    load = load_scenarios_kwh.reshape(-1)
    pv = pv_scenarios_kwh.reshape(-1)
    probabilities = np.asarray(scenario_probabilities)
    emergency_price = (
        np.asarray(price_yuan_per_kwh)
        if scenario_price_yuan_per_kwh is None
        else np.asarray(scenario_price_yuan_per_kwh)
    )
    M = storage.power_kw * DT_H

    q_slice = slice(0, horizon)
    up_slice = slice(horizon, 2 * horizon)
    down_slice = slice(2 * horizon, 3 * horizon)
    c_start = 3 * horizon
    c_slice = slice(c_start, c_start + S * horizon)
    d_slice = slice(c_slice.stop, c_slice.stop + S * horizon)
    soc_start = d_slice.stop
    soc_slice = slice(soc_start, soc_start + S * horizon)
    emergency_start = soc_slice.stop
    emergency_slice = slice(
        emergency_start,
        emergency_start + S * horizon,
    )
    curtail_start = emergency_slice.stop
    curtail_slice = slice(curtail_start, curtail_start + S * horizon)
    variable_count = curtail_slice.stop

    objective = np.zeros(variable_count)
    objective[q_slice] = price_yuan_per_kwh
    objective[up_slice] = 1.5 * price_yuan_per_kwh
    objective[down_slice] = 0.5 * price_yuan_per_kwh
    objective[emergency_slice] = (
        5.0 * emergency_price * np.repeat(probabilities, horizon)
    )

    lower = np.zeros(variable_count)
    upper = np.full(variable_count, np.inf)
    upper[c_slice] = M
    upper[d_slice] = M
    lower[soc_slice] = storage.soc_min_kwh
    upper[soc_slice] = storage.soc_max_kwh
    upper[curtail_slice] = pv
    if not adjustment_mode:
        upper[up_slice] = 0.0
        upper[down_slice] = 0.0
    else:
        upper[up_slice.start + current_intervals:up_slice.stop] = 0.0
        upper[down_slice.start + current_intervals:down_slice.stop] = 0.0
    if fixed_purchase is not None:
        executed = q_slice.start + current_intervals
        lower[q_slice.start:executed] = fixed_purchase[:current_intervals]
        upper[q_slice.start:executed] = fixed_purchase[:current_intervals]

    balance = lil_matrix((S * horizon, variable_count))
    mutual = lil_matrix((S * horizon, variable_count))
    soc = lil_matrix((S * horizon, variable_count))
    soc_rhs = np.zeros(S * horizon)
    for k in range(S):
        offset = k * horizon
        for t in range(horizon):
            row = offset + t
            balance[row, t] = 1.0
            balance[row, emergency_start + row] = 1.0
            balance[row, d_slice.start + row] = 1.0
            balance[row, c_slice.start + row] = -1.0
            balance[row, curtail_start + row] = -1.0
            mutual[row, c_slice.start + row] = 1.0
            mutual[row, d_slice.start + row] = 1.0
            soc[row, soc_start + row] = 1.0
            soc[row, c_slice.start + row] = -storage.efficiency
            soc[row, d_slice.start + row] = 1.0 / storage.efficiency
            if t > 0:
                soc[row, soc_start + row - 1] = -1.0
        soc_rhs[offset] = initial_soc_kwh

    constraints = [
        LinearConstraint(balance.tocsr(), load - pv, load - pv),
        LinearConstraint(soc.tocsr(), soc_rhs, soc_rhs),
        LinearConstraint(
            mutual.tocsr(),
            np.full(S * horizon, -np.inf),
            np.full(S * horizon, M),
        ),
    ]
    if adjustment_mode:
        deviation = lil_matrix((current_intervals, variable_count))
        for t in range(current_intervals):
            deviation[t, t] = 1.0
            deviation[t, up_slice.start + t] = -1.0
            deviation[t, down_slice.start + t] = 1.0
        constraints.append(
            LinearConstraint(
                deviation.tocsr(),
                plan_purchase[:current_intervals],
                plan_purchase[:current_intervals],
            )
        )

    result = milp(
        c=objective,
        integrality=None,
        bounds=Bounds(lower, upper),
        constraints=constraints,
        options={"time_limit": float(time_limit_s)},
    )
    solution = np.asarray(result.x)
    return {
        "purchase_kwh": _zero_small(solution[q_slice]),
        "up_kwh": _zero_small(solution[up_slice]),
        "down_kwh": _zero_small(solution[down_slice]),
        "objective_value_yuan": float(objective @ solution),
    }
```

真实负荷和真实光伏到达后，储能动作不冻结。程序逐 10 分钟比较“当前放电减少紧急购电”和“保留电量用于未来”的边际价值，只使用当前已到达的数据执行决策。

```python
# 摘自：问题三/problem3_algorithm.py

def execute_block_with_future_value(
    actual_load_kwh,
    actual_pv_kwh,
    purchase_kwh,
    load_scenarios_kwh,
    pv_scenarios_kwh,
    scenario_probabilities,
    price_yuan_per_kwh,
    storage,
    *,
    initial_soc_kwh,
    terminal_soc_value_yuan_per_kwh,
    soc_grid_points=61,
    scenario_price_yuan_per_kwh=None,
):
    horizon = len(actual_load_kwh)
    soc_grid, future_value = _build_future_value_tables(
        load_scenarios_kwh,
        pv_scenarios_kwh,
        scenario_probabilities,
        purchase_kwh,
        price_yuan_per_kwh,
        storage,
        terminal_soc_value_yuan_per_kwh=terminal_soc_value_yuan_per_kwh,
        soc_grid_points=soc_grid_points,
        scenario_price_yuan_per_kwh=scenario_price_yuan_per_kwh,
    )
    M = storage.power_kw * DT_H
    charge = np.zeros(horizon)
    discharge = np.zeros(horizon)
    emergency = np.zeros(horizon)
    curtail = np.zeros(horizon)
    soc = np.empty(horizon + 1)
    soc[0] = initial_soc_kwh

    for t in range(horizon):
        residual = (
            actual_load_kwh[t]
            - actual_pv_kwh[t]
            - purchase_kwh[t]
        )
        current_soc = float(soc[t])
        if residual <= 0.0:
            charge_value = max(
                0.0,
                min(
                    -residual,
                    M,
                    (storage.soc_max_kwh - current_soc)
                    / storage.efficiency,
                ),
            )
            discharge_value = 0.0
            emergency_value = 0.0
            curtail_value = max(0.0, -residual - charge_value)
        else:
            max_discharge = max(
                0.0,
                min(
                    residual,
                    M,
                    storage.efficiency
                    * (current_soc - storage.soc_min_kwh),
                ),
            )
            if max_discharge <= 1e-12:
                discharge_value = 0.0
            else:
                candidates = np.linspace(0.0, max_discharge, 81)
                candidate_soc = (
                    current_soc
                    - candidates / storage.efficiency
                )
                future = np.interp(
                    candidate_soc,
                    soc_grid,
                    future_value[t + 1],
                )
                objective = (
                    5.0
                    * price_yuan_per_kwh[t]
                    * (residual - candidates)
                    + future
                )
                discharge_value = float(
                    candidates[int(np.argmin(objective))]
                )
            charge_value = 0.0
            emergency_value = max(0.0, residual - discharge_value)
            curtail_value = 0.0

        charge[t] = charge_value
        discharge[t] = discharge_value
        emergency[t] = emergency_value
        curtail[t] = curtail_value
        soc[t + 1] = (
            current_soc
            + storage.efficiency * charge_value
            - discharge_value / storage.efficiency
        )
    return {
        "charge_kwh": charge,
        "discharge_kwh": discharge,
        "emergency_purchase_kwh": emergency,
        "actual_curtail_kwh": curtail,
        "soc_kwh": soc,
    }
```

实际结算按下式计算紧急购电：

\[
e_t^F=\max\left(0,L_t+c_t^F-q_t-W_t^F-d_t^F\right).
\]

多余光伏允许弃光，不向外部电网售电。

### B.4  问题四：波动电价与联合情景

问题 4-2、4-3 使用附件 4 的实时电价。为保持决策因果性，第 \(d\) 天的价格预测只使用严格早于第 \(d\) 天的已实现价格。6:00、12:00、18:00 的价格情景采用“已观察价格 + 历史同日同时段增量”，最终结算才使用当日实际实时价格。

```python
# 摘自：问题四/problem4_core.py

def build_causal_price_forecast(
    price_by_date,
    periods_per_day=PERIODS_PER_DAY,
    first_day_fallback_yuan_per_kwh=1.0,
):
    cumulative = np.zeros(periods_per_day)
    forecast = {}
    count = 0
    for current_date in sorted(price_by_date):
        actual = np.asarray(price_by_date[current_date])
        forecast[current_date] = (
            np.full(periods_per_day, first_day_fallback_yuan_per_kwh)
            if count == 0
            else cumulative / count
        )
        cumulative += actual
        count += 1
    return forecast


def build_problem4_scenario_windows(
    data,
    forecast_by_date,
    current_date,
    price_by_date,
    expected_price_yuan_per_kwh,
    fallback_load_profile_kwh,
    *,
    scenario_count=5,
    lookback_days=30,
    forecast_scale=1.0,
):
    windows = build_day_scenario_windows(
        data,
        forecast_by_date,
        current_date,
        np.asarray(expected_price_yuan_per_kwh),
        fallback_load_profile_kwh,
        scenario_count=scenario_count,
        lookback_days=lookback_days,
        window_days=1,
        forecast_scale=forecast_scale,
    )
    actual_price = (
        data[data["日期"].dt.date == current_date]
        .sort_values("时段序号")["电价_元每kWh"]
        .to_numpy()
    )
    for start_hour, window in windows.items():
        S, horizon = np.asarray(window["load_kwh"]).shape
        current_intervals = int(window["current_intervals"])
        selected_dates = tuple(window["selected_error_dates"])
        if not selected_dates:
            selected_dates = (current_date,) * S
        scenario_prices = np.empty((S, horizon))

        for k, history_date in enumerate(selected_dates):
            history_price = np.asarray(price_by_date[history_date])
            if start_hour == 0:
                current_curve = history_price[:current_intervals].copy()
            else:
                start = start_hour * 6
                reference_actual = float(actual_price[start - 1])
                reference_history = float(history_price[start - 1])
                current_curve = (
                    reference_actual
                    + history_price[start:]
                    - reference_history
                )
            scenario_prices[k] = np.maximum(current_curve, 1e-6)

        probabilities = np.asarray(window["probabilities"])
        window["price_scenarios_yuan_per_kwh"] = scenario_prices
        window["price_yuan_per_kwh"] = probabilities @ scenario_prices
    return windows
```
